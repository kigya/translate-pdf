cmake_minimum_required(VERSION 3.22)
project(appmodules LANGUAGES C CXX)
option(TRANSLATEPDF_HOST_COVERAGE "Instrument owned C++ host code with LLVM coverage" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(DEPS "${PROJECT_ROOT}/.local-private/deps")
if(ANDROID)
 set(PDFIUM_PLATFORM android-arm64)
elseif(IOS)
 if(CMAKE_OSX_SYSROOT MATCHES "simulator")
  set(PDFIUM_PLATFORM ios-simulator-arm64)
 else()
  set(PDFIUM_PLATFORM ios-device-arm64)
 endif()
else()
 set(PDFIUM_PLATFORM mac-arm64)
endif()
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/llama.cpp" "${CMAKE_CURRENT_BINARY_DIR}/llama")
# Inference kernels must remain optimized even in the debuggable application.
foreach(kernel llama ggml ggml-base ggml-cpu)
 if(TARGET ${kernel})
  target_compile_options(${kernel} PRIVATE $<$<CONFIG:Debug>:-O3>)
 endif()
endforeach()
set(PDFium_DIR "${DEPS}/pdfium/${PDFIUM_PLATFORM}")
set(PDFium_INCLUDE_DIR "${PDFium_DIR}/include" CACHE PATH "Pinned PDFium headers" FORCE)
if(ANDROID)
 set(PDFium_LIBRARY "${PDFium_DIR}/lib/libpdfium.so" CACHE FILEPATH "Pinned PDFium library" FORCE)
else()
 set(PDFium_LIBRARY "${PDFium_DIR}/lib/libpdfium.dylib" CACHE FILEPATH "Pinned PDFium library" FORCE)
endif()
find_package(PDFium CONFIG REQUIRED)
if(ANDROID)
 if(DEFINED REACT_ANDROID_DIR)
  add_subdirectory(react)
 endif()
 add_library(translatepdf SHARED core.cpp docx.cpp vendor/miniz.c vendor/miniz_zip.c vendor/miniz_tdef.c vendor/miniz_tinfl.c vendor/pugixml.cpp android_bridge.cpp)
 target_link_libraries(translatepdf PRIVATE llama pdfium android jnigraphics log)
 target_link_options(translatepdf PRIVATE "-Wl,-z,max-page-size=16384")
else()
 add_library(translatepdf STATIC core.cpp docx.cpp vendor/miniz.c vendor/miniz_zip.c vendor/miniz_tdef.c vendor/miniz_tinfl.c vendor/pugixml.cpp)
 target_link_libraries(translatepdf PUBLIC llama pdfium)
 add_executable(native-smoke smoke.cpp)
 target_link_libraries(native-smoke PRIVATE translatepdf)
 target_include_directories(native-smoke PRIVATE "${DEPS}/include")
 if(NOT IOS)
  enable_testing()
  add_executable(translation-guard-test guard_test.cpp)
  add_test(NAME translation-consistency COMMAND translation-guard-test)
  # Compile the production JNI adapter on macOS with the host JDK's real JNI
  # declarations and a narrow AndroidBitmap boundary shim. The test drives the
  # exported entrypoints and native PDFium/core implementation; it does not
  # replace Android device validation.
  find_package(JNI REQUIRED)
  add_executable(jni-bridge-test tests/jni_bridge_test.cpp android_bridge.cpp)
  target_include_directories(jni-bridge-test PRIVATE
   "${JNI_INCLUDE_DIRS}" "${CMAKE_CURRENT_LIST_DIR}/tests/jni_shim" "${DEPS}/include")
  target_link_libraries(jni-bridge-test PRIVATE translatepdf)
  add_test(NAME android-jni-bridge COMMAND jni-bridge-test)
  add_executable(core-detail-test tests/core_detail_test.cpp)
  target_include_directories(core-detail-test PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${DEPS}/include")
  target_link_libraries(core-detail-test PRIVATE translatepdf)
  add_test(NAME core-font-unicode COMMAND core-detail-test)
  add_executable(core-fault-test tests/core_fault_test.cpp)
  target_include_directories(core-fault-test PRIVATE "${DEPS}/include")
  target_compile_definitions(core-fault-test PRIVATE PROJECT_ROOT="${PROJECT_ROOT}")
  target_link_libraries(core-fault-test PRIVATE translatepdf)
  add_test(NAME core-boundary-faults COMMAND core-fault-test)
  add_executable(docx-core-test tests/docx_core_test.cpp)
  target_link_libraries(docx-core-test PRIVATE translatepdf)
  target_include_directories(docx-core-test PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMAKE_CURRENT_LIST_DIR}/vendor" "${DEPS}/include")
  add_test(NAME docx-core-roundtrip-and-safety COMMAND docx-core-test)
  add_executable(docx-fault-test tests/docx_fault_test.cpp)
  target_link_libraries(docx-fault-test PRIVATE translatepdf)
  target_include_directories(docx-fault-test PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMAKE_CURRENT_LIST_DIR}/vendor" "${DEPS}/include")
  add_test(NAME docx-zip-boundary-faults COMMAND docx-fault-test)
  if(APPLE)
   # CTest launchers may sanitize inherited DYLD variables on macOS. Supply
   # the pinned PDFium location directly to the unsigned host test children.
   set_tests_properties(android-jni-bridge core-font-unicode core-boundary-faults docx-core-roundtrip-and-safety docx-zip-boundary-faults
    PROPERTIES ENVIRONMENT "DYLD_LIBRARY_PATH=${PDFium_DIR}/lib")
  endif()
  if(TRANSLATEPDF_HOST_COVERAGE)
   foreach(owned translatepdf translation-guard-test jni-bridge-test docx-core-test docx-fault-test)
    target_compile_options(${owned} PRIVATE -fprofile-instr-generate -fcoverage-mapping)
   endforeach()
   target_link_options(native-smoke PRIVATE -fprofile-instr-generate)
   target_link_options(translation-guard-test PRIVATE -fprofile-instr-generate)
   target_link_options(jni-bridge-test PRIVATE -fprofile-instr-generate)
   target_link_options(core-detail-test PRIVATE -fprofile-instr-generate)
   target_link_options(core-fault-test PRIVATE -fprofile-instr-generate)
   target_link_options(docx-core-test PRIVATE -fprofile-instr-generate)
   target_link_options(docx-fault-test PRIVATE -fprofile-instr-generate)
  endif()
 endif()
endif()
target_include_directories(translatepdf PUBLIC "${CMAKE_CURRENT_LIST_DIR}" PRIVATE "${DEPS}/include" "${CMAKE_CURRENT_LIST_DIR}/vendor")
